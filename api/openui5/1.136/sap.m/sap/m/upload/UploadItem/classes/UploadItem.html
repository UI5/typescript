Class: UploadItem

import UploadItem from "sap/m/upload/UploadItem";

sap.m.upload.UploadItem represents one item to be uploaded using the sap.m.plugins.UploadSetwithTable UploadSetwithTable

Note: This element should only be used within the sap.m.plugins.UploadSetwithTable UploadSetwithTable plugin.
